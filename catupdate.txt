def answer(

    prompt: str,

    max_tokens: int = 768,

    images: list[str] | tuple[str, ...] | None = None,

) -> tuple[str, str, dict]:

    image_tuple = tuple(images or ())

    ctx = RequestContext(prompt=prompt, max_tokens=max_tokens, images=image_tuple)



    decision = gemma.decide(ctx.prompt, has_images=bool(ctx.images))

    profile = vram_guard.profile_for_decision(decision.worker)

    vram_guard.assert_profile_safe(profile)



    if decision.worker == "qwen":

        result = gemmacodercomms.run(ctx.prompt, decision.task, ctx.max_tokens)

    elif decision.worker == "safetensor":

        result = gemmaimagingcomms.run(ctx.prompt, decision.task)

    elif decision.worker == "vision":

        result = gemmavisioncomms.run(ctx.prompt, list(ctx.images), ctx.max_tokens)

    else:

        result = gemma.write(decision.task, ctx.max_tokens)



    metadata = result.metadata.copy()

    metadata.update(

        {

            "decision": decision.__dict__,

            "ok": result.ok,

            "error": result.error,

            "vram_profile": profile.as_dict(),

            "parameter_governor": vram_guard.parameter_governor(),

        }

    )



    return result.worker, result.content, metadata

def decide(prompt: str, has_images: bool = False) -> WorkerDecision:

    p = prompt.lower()

    if has_images:

        return WorkerDecision(worker="vision", task=prompt, kind="vision", reason="image_input")

    if any(x in p for x in ("draw", "image", "picture", "photo", "render", "generate a logo", "stable", "safetensor")):

        return WorkerDecision(worker="safetensor", task=prompt, kind="image", reason="image_generation_request")

    if any(x in p for x in ("code", "script", "python", "bash", "rust", "cpp", "c++", "debug", "repo", "bitbake", ".bb", "kernel", "cuda")):

        return WorkerDecision(worker="qwen", task=prompt, kind="code", reason="code_or_build_request")

    return WorkerDecision(worker="gemma", task=prompt, kind="chat", reason="general_chat")





def write(prompt: str, max_tokens: int = 768) -> WorkerResult:

    vram_guard.assert_safe_for_job(NAME)

    with flow_control.controller.job("text"):

        kv_cache.observe(prompt, NAME)

        content = text_engine.generate(

            name=NAME,

            model_ref=MODELS.gemma,

            system=SYSTEM,

            prompt=prompt,

            max_new_tokens=max_tokens,

            temperature=SETTINGS.temperature_chat,

        )

    return WorkerResult(worker=NAME, content=content, metadata={"backend": text_engine.backend_name()})





def final_check(original: str, draft: str, max_tokens: int = 384) -> WorkerResult:

    prompt = (

        "Review the specialist answer for correctness, safety, and directness. "

        "Return the final answer only.\n\n"

        f"Original request:\n{original}\n\nSpecialist draft:\n{draft}"

    )

    try:

        return write(prompt, max_tokens=max_tokens)

        return WorkerResult(worker=NAME, content=draft, metadata={"final_check": "skipped_due_to_gemma_unavailable"})

def _save_data_url(data_url: str) -> str:

    header, payload = data_url.split(",", 1)

    ext = "png"

    if "image/jpeg" in header:

        ext = "jpg"

    elif "image/webp" in header:

        ext = "webp"

    path = UPLOADS / f"input_{uuid.uuid4().hex}.{ext}"

    path.write_bytes(base64.b64decode(payload))

    return str(path)





def _resolve_image_url(url: str) -> str:

    url = url.strip()

    if url.startswith("data:image/"):

        return _save_data_url(url)

    parsed = urlparse(url)

    if parsed.scheme in {"", "file"}:

        return str(Path(parsed.path or url).expanduser())

    # Do not fetch remote images inside production server. Download explicitly before submit.

    raise HTTPException(400, f"remote_image_url_not_allowed:{parsed.scheme}")





def extract_user_payload(messages: list[ChatMessage]) -> tuple[str, list[str]]:

    prompt_parts: list[str] = []

    image_paths: list[str] = []

    for message in reversed(messages):

        if message.role != "user":

            continue

        content = message.content

        if isinstance(content, str):

            prompt_parts.append(content.strip())

        elif isinstance(content, list):

            for item in content:

                if not isinstance(item, dict):

                    continue

                typ = item.get("type")

                if typ in {"text", "input_text"}:

                    prompt_parts.append(str(item.get("text", "")).strip())

                elif typ in {"image_url", "input_image"}:

                    raw = item.get("image_url") or item.get("image") or item.get("url")

                    if isinstance(raw, dict):

                        raw = raw.get("url")

                    if raw:

                        image_paths.append(_resolve_image_url(str(raw)))

        elif isinstance(content, dict):

            text = content.get("text") or content.get("prompt")

            if text:

                prompt_parts.append(str(text).strip())

        break

    prompt = "\n".join(x for x in prompt_parts if x).strip()

    return prompt, image_paths

    if req.stream:

        raise HTTPException(400, "streaming_not_enabled_for_inprocess_backend")

    prompt, images = extract_user_payload(req.messages)

    if not prompt and not images:

        raise HTTPException(400, "missing_user_prompt")

    if not prompt and images:

        prompt = "Describe this image and answer any obvious question about it."

    started = time.monotonic()

    try:

        worker, content, metadata = await run_in_threadpool(

            symoneural.answer,

            prompt,

            req.max_tokens or SETTINGS.max_new_tokens_default,

            images,

        )

    except RuntimeError as exc:

        raise HTTPException(503, str(exc)) from exc

    return {

        "id": f"symon-chatcmpl-{uuid.uuid4().hex[:12]}",

        "object": "chat.completion",

        "created": int(time.time()),

        "model": worker,

        "choices": [{

            "index": 0,

            "message": {"role": "assistant", "content": content},

            "finish_reason": "stop",

        }],

        "usage": {"prompt_tokens": None, "completion_tokens": None, "total_tokens": None},

        "symoneural": {

            "latency_ms": round((time.monotonic() - started) * 1000, 2),

            "worker": worker,

            "images": len(images),

            "client": request.client.host if request.client else None,

            **metadata,

        },

    }

def bool_from_env(name: str, default: bool) -> bool:

    raw = os.environ.get(name)

def load_env_file() -> bool:

    env_path = ROOT / ".env"

    if not env_path.is_file():

        return False



    key_re = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*$")



    for raw in env_path.read_text(errors="ignore").splitlines():

        line = raw.strip()



        if not line or line.startswith("#"):

            continue



        if line.startswith("export "):

            line = line[len("export ") :].strip()



        if "=" not in line:

            continue



        key, value = line.split("=", 1)

        key = key.strip()

        value = value.strip()



        if not key_re.match(key):

            continue



        if len(value) >= 2 and value[0] == value[-1] and value[0] in {"'", '"'}:

            value = value[1:-1]



        os.environ.setdefault(key, value)



    return True





def env_first(names: list[str], fallback: str) -> str:

    for name in names:

        value = os.environ.get(name)

        if value:

            return value

    return fallback

def apply_model_refs_to_config(config_module: Any, models_obj: Any, refs: dict[str, str]) -> Any:

    updates = {

        "gemma": refs["gemma"],

        "qwen": refs["qwen"],

        "vision": refs["vision"],

        "stable": refs["stable"],

    }



    if is_dataclass(models_obj):

        try:

            new_models = replace(models_obj, **updates)

            setattr(config_module, "MODELS", new_models)

            return new_models

        except Exception:

            pass



    if hasattr(models_obj, "model_copy"):

        try:

            new_models = models_obj.model_copy(update=updates)

            setattr(config_module, "MODELS", new_models)

            return new_models

        except Exception:

            pass



    for key, value in updates.items():

        try:

            setattr(models_obj, key, value)

        except Exception:

            pass



    try:

        setattr(config_module, "MODELS", models_obj)

    except Exception:

        pass



    return models_obj





def detail_ok(detail: Any) -> bool:

    if isinstance(detail, dict):

        return bool(detail.get("ok"))

    return bool(detail)





def call_ready_detail(fn: Any, *args: Any, **kwargs: Any) -> dict[str, Any]:

    try:

        detail = fn(*args, **kwargs)

        if isinstance(detail, dict):

            return detail

        return {"ok": bool(detail), "detail": detail}

    except Exception as exc:

        return {"ok": False, "error": repr(exc)}





def _contains_all(body: str, tokens: tuple[str, ...]) -> bool:

    return all(token in body for token in tokens)

def _shares_close(values: list[float], expected: list[float], tolerance: float = 0.001) -> bool:

    if len(values) != len(expected):

        return False



    return all(abs(a - b) <= tolerance for a, b in zip(values, expected, strict=True))

            if detail:
